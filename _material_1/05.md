# 5. Tietorakenteet

## Rakenteinen tietotyyppi

Avainsana `struct` määrittelee rakenteisen tietotyypin, jonka sisällä on nimettyjä kenttiä. Esimerkiksi seuraava tietotyyppi `Position` sisältää kentät `x` ja `y`:

```c
struct Position {
    int x;
    int y;
}
```

Seuraava koodi määrittelee muuttujan `pos`, jonka tyyppinä on `struct Position`.

```c
struct Position pos = {2, 3};

printf("%d %d\n", pos.x, pos.y);
```

Rakenteisen tietotyypin kenttien arvot voidaan asettaa muuttujan luonnissa samassa järjestyksessä kuin ne on annettu tyypin määrittelyssä. Esimerkiksi yllä oleva koodi asettaa kentän `x` arvoksi 2 ja kentän `y` arvoksi 3.

Muuttujan kenttien arvoihin voidaan viitata `.`-syntaksin avulla, kuten yllä olevassa koodissa `pos.x` ja `pos.y`.

### Sisällön kopiointi

Rakenteinen tietotyyppi muistuttaa alkeistyyppiä (kuten `int`), koska muuttujan sisältö kopioidaan sijoituksessa ja funktion parametrina. Esimerkiksi:

```c
struct Position {
    int x;
    int y;
};

void test(struct Position pos) {
    pos.x++;
    pos.y++;
    printf("%d %d\n", pos.x, pos.y); // 3 4
}

int main(void) {
    struct Position pos = {2, 3};
    test(pos);
    printf("%d %d\n", pos.x, pos.y); // 2 3
}
```

Tässä funktion `test` parametrin `pos` sisältö kopioidaan eivätkä funktion tekemät muutokset parametrin kenttiin vaikuta pääohjelmaan.

Hieman yllättäen muuttujan sisältö kopioidaan jopa siinä tapauksessa, että rakenteisen tietotyypin kenttänä on taulukko:

```c
struct Array {
    int numbers[4];
};

void test(struct Array array) {
    array.numbers[0] = 5;
    printf("%d\n", array.numbers[0]); // 5
}

int main(void) {
    struct Array array = { {1, 2, 3, 4} };
    test(array);
    printf("%d\n", array.numbers[0]); // 1
}
```

Siis jos taulukko on sellaisenaan funktion parametrina, sen sisältöä ei kopioida vaan taulukko välitetään osoittimena, mutta jos taulukko on rakenteisen tietotyypin sisällä, sen sisältö kopioidaan.

### Osoitinsyntaksi

Kun osoitin osoittaa rakenteisen tietotyypin arvoon, kentän sisältöön viittaamiseen on erityinen syntaksi `pointer->field`, joka tarkoittaa samaa kuin `(*pointer).field`. Seuraavat kaksi koodia toimivat samalla tavalla:

```c
struct Position pos = {2, 3};
struct Position *pointer = &pos;

printf("%d %d\n", (*pointer).x, (*pointer).y); // 2 3
```

```c
struct Position pos = {2, 3};
struct Position *pointer = &pos;

printf("%d %d\n", pointer->x, pointer->y); // 2 3
```

### Lyhempi tyypin nimi

Rakenteisen tietotyypin käyttämisessä voi olla vaivalloista, että tyypin nimeen tulee liittää sana `struct` esimerkiksi muuttujan määrittelyssä:

```c
struct Position pos = {2, 3};
```

Tämä on kuitenkin mahdollista kiertää seuraavasti:

```c
typedef struct Position Position;

struct Position {
    int x;
    int y;
};
```

Tässä `typedef`-komento määrittelee, että tyyppi `Position` tarkoittaa samaa kuin `struct Position`. Nyt muuttuja voidaan määritellä näin:

```c
Position pos = {2, 3};
```

## Taulukkolista

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct ArrayList {
    int size;
    int capacity;
    int* items;
} ArrayList;

ArrayList* array_list_create() {
    ArrayList* list = malloc(sizeof(ArrayList));
    list->size = 0;
    list->capacity = 4;
    list->items = malloc(sizeof(int) * list->capacity);
    return list;
}

void array_list_free(ArrayList* list) {
    free(list->items);
    free(list);
}

void array_list_insert(ArrayList* list, int value) {
    if (list->capacity == list->size) {
        list->capacity *= 2;
        list->items = realloc(list->items, sizeof(int) * list->capacity);
    }
    list->items[list->size] = value;
    list->size++;
}

void array_list_print(ArrayList* list) {
    printf("[");
    for (int i = 0; i < list->size; i++) {
        if (i != 0) printf(", ");
        printf("%d", list->items[i]);
    }
    printf("]\n");
}

int main(void) {
    ArrayList* list = array_list_create();
    for (int i = 1; i <= 10; i++) {
        array_list_insert(list, i);
        array_list_print(list);
    }
    array_list_free(list);
}
```

## Linkitetty lista

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct ListNode {
    struct ListNode *prev, *next;
    int value;
} ListNode;

typedef struct LinkedList {
    int size;
    ListNode *first, *last;
} LinkedList;

LinkedList* linked_list_create() {
    LinkedList* list = malloc(sizeof(LinkedList));
    list->size = 0;
    list->first = NULL;
    list->last = NULL;
    return list;
}

void linked_list_free(LinkedList* list) {
    ListNode *node = list->first;
    while (node) {
        ListNode *next = node->next;
        free(node);
        node = next;
    }
    free(list);
}

void linked_list_insert(LinkedList* list, int value) {
    if (list->size == 0) {
        ListNode* node = malloc(sizeof(ListNode));
        node->prev = NULL;
        node->next = NULL;
        node->value = value;
        list->first = node;
        list->last = node;
        list->size = 1;
    } else {
        ListNode* node = malloc(sizeof(ListNode));
        node->prev = list->last;
        list->last->next = node;
        node->next = NULL;
        node->value = value;
        list->last = node;
        list->size++;
    }
}

void linked_list_print(LinkedList* list) {
    printf("[");
    ListNode* node = list->first;
    int start = 1;
    while (node) {
        if (!start) printf(", ");
        start = 0;
        printf("%d", node->value);
        node = node->next;
    }
    printf("]\n");
}

int main(void) {
    LinkedList* list = linked_list_create();
    for (int i = 1; i <= 10; i++) {
        linked_list_insert(list, i);
        linked_list_print(list);
    }
    linked_list_free(list);
}
```

## Hajautustaulu

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct ListNode {
    struct ListNode *prev, *next;
    char *value;
} ListNode;

typedef struct LinkedList {
    int size;
    ListNode *first, *last;
} LinkedList;

LinkedList* linked_list_create() {
    LinkedList* list = malloc(sizeof(LinkedList));
    list->size = 0;
    list->first = NULL;
    list->last = NULL;
    return list;
}

void linked_list_free(LinkedList* list) {
    ListNode *node = list->first;
    while (node) {
        ListNode *next = node->next;
        free(node);
        node = next;
    }
    free(list);
}

void linked_list_insert(LinkedList* list, char *value) {
    if (list->size == 0) {
        ListNode* node = malloc(sizeof(ListNode));
        node->prev = NULL;
        node->next = NULL;
        node->value = value;
        list->first = node;
        list->last = node;
        list->size = 1;
    } else {
        ListNode* node = malloc(sizeof(ListNode));
        node->prev = list->last;
        list->last->next = node;
        node->next = NULL;
        node->value = value;
        list->last = node;
        list->size++;
    }
}

int linked_list_check(LinkedList* list, char *string) {
    ListNode *node = list->first;
    while (node) {
        if (strcmp(node->value, string) == 0) {
            return 1;
        }
        node = node->next;
    }
    return 0;
}

typedef struct HashTable {
    LinkedList **lists;
    int size;
} HashTable;

HashTable* hash_table_create() {
    HashTable *table = malloc(sizeof(HashTable));
    table->size = 1000;
    table->lists = malloc(sizeof(LinkedList*) * table->size);
    for (int i = 0; i < table->size; i++) {
        table->lists[i] = linked_list_create();
    }
    return table;
}

void hash_table_free(HashTable *table) {
    for (int i = 0; i < table->size; i++) {
        linked_list_free(table->lists[i]);
    }
    free(table->lists);
    free(table);
}

unsigned long hash_value(char *string) {
    unsigned long value = 0;
    for (int i = 0; string[i]; i++) {
        value = value * 41 + string[i];
    }
    return value;
}

void hash_table_insert(HashTable *table, char *string) {
    int pos = hash_value(string) % table->size;
    if (linked_list_check(table->lists[pos], string)) {
        return;
    }
    linked_list_insert(table->lists[pos], string);
}

int hash_table_check(HashTable *table, char *string) {
    int pos = hash_value(string) % table->size;
    return linked_list_check(table->lists[pos], string);
}

int main(void) {
    HashTable *table = hash_table_create();
    hash_table_insert(table, "apina");
    hash_table_insert(table, "banaani");
    hash_table_insert(table, "cembalo");
    printf("%d\n", hash_table_check(table, "banaani"));
    printf("%d\n", hash_table_check(table, "faarao"));
    hash_table_free(table);
}
```
